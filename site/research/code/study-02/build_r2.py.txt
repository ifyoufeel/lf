def b64png(arr8):
    im = Image.fromarray(arr8.astype("uint8"), mode="L"); bio = io.BytesIO(); im.save(bio, format="PNG", optimize=True)
    return "data:image/png;base64," + base64.b64encode(bio.getvalue()).decode()

def down(a, n=150, mode="mean"):
    H = a.shape[0]; k = H // n
    a = a[:k * n, :k * n].astype("float32").reshape(n, k, n, k)
    if mode == "mean":
        with np.errstate(all="ignore"): return np.nanmean(a, axis=(1, 3))
    return a[:, k // 2, :, k // 2]

def ndvi8(a):
    a = np.where(np.isfinite(a), a, 0)
    return np.clip(a / 0.9 * 255, 0, 255)

def et0_pm(df, lat):
    Tmax, Tmin = df.T2M_MAX.values, df.T2M_MIN.values; T = (Tmax + Tmin) / 2; P = df.PS.values / 10.0
    gamma = 0.000665 * P
    es = (0.6108 * np.exp(17.27 * Tmax / (Tmax + 237.3)) + 0.6108 * np.exp(17.27 * Tmin / (Tmin + 237.3))) / 2
    ea = 0.6108 * np.exp(17.27 * df.T2MDEW.values / (df.T2MDEW.values + 237.3))
    delta = 4098 * (0.6108 * np.exp(17.27 * T / (T + 237.3))) / (T + 237.3) ** 2
    u2 = df.WS2M.values; Rs = df.RS.values * 0.0864
    J = pd.to_datetime(df.index).day_of_year.values; phi = np.deg2rad(lat)
    dr = 1 + 0.033 * np.cos(2 * np.pi * J / 365); dec = 0.409 * np.sin(2 * np.pi * J / 365 - 1.39)
    ws = np.arccos(np.clip(-np.tan(phi) * np.tan(dec), -1, 1))
    Ra = 24 * 60 / np.pi * 0.0820 * dr * (ws * np.sin(phi) * np.sin(dec) + np.cos(phi) * np.cos(dec) * np.sin(ws))
    alt = 44331.5 * (1 - (P / 101.325) ** 0.190263); Rso = (0.75 + 2e-5 * alt) * Ra
    Rnl = 4.903e-9 * (((Tmax + 273.16) ** 4 + (Tmin + 273.16) ** 4) / 2) * (0.34 - 0.14 * np.sqrt(np.maximum(ea, 0))) * (1.35 * np.clip(Rs / np.maximum(Rso, 1e-6), 0.25, 1.0) - 0.35)
    et0 = (0.408 * delta * (0.77 * Rs - Rnl) + gamma * 900 / (T + 273) * u2 * (es - ea)) / (delta + gamma * (1 + 0.34 * u2))
    hs = 0.0023 * (Ra * 0.408) * (T + 17.8) * np.sqrt(np.maximum(Tmax - Tmin, 0))
    return pd.DataFrame(dict(et0=et0, et0_hs=hs, vpd=es - ea, rs=Rs, dli=Rs * 2.04, tmax=Tmax, tmin=Tmin), index=df.index)

def water(curve_doy, curve_ndvi, wx, lat, kc_mid, year=2026, name=""):
    days = pd.date_range(f"{year}-01-01", f"{year}-10-15")
    days = days[(days.day_of_year >= int(min(curve_doy))) & (days.day_of_year <= int(max(curve_doy)))]
    nd = np.interp(days.day_of_year, curve_doy, curve_ndvi)
    present = np.where(nd >= 0.30)[0]
    if not len(present): return None
    i0 = max(present[0] - 15, 0); i1 = present[-1]
    days = days[i0:i1 + 1]; nd = nd[i0:i1 + 1]
    e = et0_pm(wx.dropna(subset=["T2M_MAX", "RS"]), lat).reindex(days)
    kc = 0.30 + (kc_mid - 0.30) * np.clip((nd - 0.15) / (0.85 - 0.15), 0, 1)
    etc = kc * e.et0.values; etc_hs = kc * e.et0_hs.values
    rain = wx.reindex(days).IMERG.fillna(wx.reindex(days).PRECTOTCORR).values
    df = pd.DataFrame(dict(etc=etc, etc_hs=etc_hs, rain=rain), index=days)
    blk = df.resample("10D").sum()
    deficit = (blk.etc - 0.8 * blk.rain).clip(lower=0).sum(); deficit_hs = (blk.etc_hs - 0.8 * blk.rain).clip(lower=0).sum()
    return dict(name=name, start=str(days[0].date()), end=str(days[-1].date()), days=len(days), kc_mid=kc_mid,
                et0=float(np.nansum(e.et0)), et0_hs=float(np.nansum(e.et0_hs)), etc=float(np.nansum(etc)), etc_hs=float(np.nansum(etc_hs)),
                rain=float(np.nansum(rain)), net_irrigation=float(deficit), net_irrigation_hs=float(deficit_hs),
                peak_etc=float(np.nanmax(etc)), mean_etc_peak_month=float(df.etc.resample("MS").mean().max()),
                daily=[dict(d=str(d.date()), etc=round(float(a), 2), rain=round(float(r), 1), ndvi=round(float(n), 3), et0=round(float(z), 2))
                       for d, a, r, n, z in zip(days, etc, rain, nd, e.et0.values)])

def _clean(o):  # browsers reject NaN in JSON
    if isinstance(o, float): return None if (o != o or o in (float("inf"), float("-inf"))) else o
    if isinstance(o, dict): return {k: _clean(v) for k, v in o.items()}
    if isinstance(o, list): return [_clean(v) for v in o]
    return o
