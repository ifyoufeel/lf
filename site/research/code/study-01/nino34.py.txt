"""Nino-3.4 and tropical-mean SST from NOAA OISST v2.1 daily files on AWS (anonymous HTTPS).
Samples three days per month (5th, 15th, 25th; the latest available day when the month is incomplete)."""
import re, sys, json, urllib.request, concurrent.futures as cf
import numpy as np, netCDF4

B = "https://noaa-cdr-sea-surface-temp-optimum-interpolation-pds.s3.amazonaws.com/"
OUT = sys.argv[1]

def get(u, tries=4):
    for i in range(tries):
        try:
            with urllib.request.urlopen(u, timeout=120) as r:
                return r.read()
        except Exception as e:
            if i == tries - 1:
                raise
    return None

def month_keys(ym):
    s = get(B + f"?list-type=2&prefix=data/v2.1/avhrr/{ym}/").decode()
    return re.findall(r"<Key>([^<]*)</Key>", s)

def pick(keys, days=(5, 15, 25)):
    by_day = {}
    for k in keys:
        m = re.search(r"\.(\d{8})(_preliminary)?\.nc$", k)
        if m:
            by_day[int(m.group(1)[6:])] = k
    if not by_day:
        return []
    out = []
    for d in days:
        if d in by_day:
            out.append(by_day[d])
        else:
            # nearest available day not after the month's last available day
            cand = sorted(by_day, key=lambda x: abs(x - d))
            out.append(by_day[cand[0]])
    return sorted(set(out))

W = None
import threading
LOCK = threading.Lock()
def fetch(key):
    return key, get(B + key)

def stats(key, raw):
    ds = netCDF4.Dataset("mem", memory=raw)
    sst = ds.variables["sst"][0, 0, :, :]  # masked array (lat, lon), lat -89.875..89.875, lon 0.125..359.875
    lat = ds.variables["lat"][:]
    lon = ds.variables["lon"][:]
    ds.close()
    w = np.cos(np.deg2rad(lat))[:, None] * np.ones((1, lon.size))
    def boxmean(la0, la1, lo0, lo1):
        mi = (lat >= la0) & (lat <= la1)
        mj = (lon >= lo0) & (lon <= lo1)
        sub = sst[np.ix_(mi, mj)]
        ww = w[np.ix_(mi, mj)]
        ww = np.where(np.ma.getmaskarray(sub), 0, ww)
        return float((np.ma.filled(sub, 0) * ww).sum() / ww.sum())
    n34 = boxmean(-5, 5, 190, 240)
    trop = boxmean(-20, 20, 0, 360)
    return key, n34, trop

def main():
    months = [f"{y}{m:02d}" for y in range(1982, 2027) for m in range(1, 13) if (y, m) <= (2026, 10)]
    with cf.ThreadPoolExecutor(16) as ex:
        lists = dict(zip(months, ex.map(month_keys, months)))
    keys = [k for ym in months for k in pick(lists[ym])]
    print("files", len(keys), file=sys.stderr)
    rows = []
    with cf.ThreadPoolExecutor(12) as ex:
        futs = [ex.submit(fetch, k) for k in keys]
        for i, f in enumerate(cf.as_completed(futs)):
            k, raw = f.result()
            r = stats(k, raw)  # netCDF4/HDF5 is not thread-safe: parse in this thread only
            rows.append(r)
            if i % 100 == 0:
                print(i, r, file=sys.stderr, flush=True)
    with open(OUT, "w") as f:
        f.write("date,file,nino34_sst,tropical_sst\n")
        for k, n, t in sorted(rows):
            d = re.search(r"\.(\d{8})(_preliminary)?\.nc$", k).group(1)
            f.write(f"{d[:4]}-{d[4:6]}-{d[6:]},{k.split('/')[-1]},{n:.4f},{t:.4f}\n")

if __name__ == "__main__":
    main()
