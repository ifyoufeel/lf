"""October 2026 SST anomaly field (2-degree grid) from OISST v2.1: Oct 2026 (5th, 15th, 24th) vs Oct 1991-2020 (5th, 15th, 25th)."""
import sys, json, urllib.request, concurrent.futures as cf
import numpy as np, netCDF4
B = "https://noaa-cdr-sea-surface-temp-optimum-interpolation-pds.s3.amazonaws.com/data/v2.1/avhrr/"
def get(u):
    for i in range(4):
        try:
            with urllib.request.urlopen(u, timeout=120) as r: return r.read()
        except Exception:
            if i == 3: raise
def name(y, d, prelim=False):
    return f"{y}10/oisst-avhrr-v02r01.{y}10{d:02d}{'_preliminary' if prelim else ''}.nc"
clim_keys = [name(y, d) for y in range(1991, 2021) for d in (5, 15, 25)]
cur_keys = [name(2026, d, True) for d in (5, 15, 24)]
def coarse(raw):
    ds = netCDF4.Dataset("m", memory=raw); sst = ds.variables["sst"][0, 0].filled(np.nan); ds.close()
    a = sst.reshape(90, 8, 180, 8)  # 0.25 deg -> 2 deg
    with np.errstate(all="ignore"):
        return np.nanmean(a, axis=(1, 3))
with cf.ThreadPoolExecutor(12) as ex:
    futs = {ex.submit(get, B + k): k for k in clim_keys + cur_keys}
    clim, cur = [], []
    for f in cf.as_completed(futs):
        k = futs[f]; arr = coarse(f.result())
        (cur if "2026" in k[:4] else clim).append(arr)
clim = np.nanmean(np.stack(clim), 0); cur = np.nanmean(np.stack(cur), 0)
anom = cur - clim
lat = (np.arange(90) * 2 - 89).tolist(); lon = (np.arange(180) * 2 + 1).tolist()
json.dump(dict(lat=lat, lon=lon, anom=[[None if not np.isfinite(v) else round(float(v), 2) for v in row] for row in anom]), open(sys.argv[1], "w"))
print("max", np.nanmax(anom), "nino34 box mean", np.nanmean(anom[42:47, 95:120]))
